\begin{tikzpicture}[

    >=Latex,
    node distance=0.7cm and 1.0cm,
    inputblock/.style={
        draw=black,
        fill=red!12,
        rounded corners=2pt,
        minimum width=2.8cm,
        minimum height=0.9cm,
        align=center
    },
    convblock/.style={
        draw=blue!70!black,
        fill=blue!15,
        rounded corners=2pt,
        minimum width=4.0cm,
        minimum height=1.0cm,
        align=center,
        thick
    },
    resblock/.style={
        draw=black!70,
        fill=gray!15,
        rounded corners=3pt,
        minimum width=4.3cm,
        minimum height=1.3cm,
        align=center,
        thick
    },
    opblock/.style={
        draw=black!70,
        fill=yellow!20,
        rounded corners=2pt,
        minimum width=3.5cm,
        minimum height=0.8cm,
        align=center
    },
    skipblock/.style={
        draw=black!70,
        fill=green!12,
        rounded corners=2pt,
        minimum width=3.5cm,
        minimum height=0.8cm,
        align=center
    },
    shapetext/.style={
        font=\scriptsize,
        align=center
    },
    mainarrow/.style={
        ->,
        thick
    },
    skiparrow/.style={
        ->,
        thick,
        dashed
    },
    projblock/.style={
        draw=purple!70!black,
        fill=purple!12,
        rounded corners=2pt,
        minimum width=3.7cm,
        minimum height=0.8cm,
        align=center
    },
    sidearrow/.style={
        ->,
        thick,
        dashed
    }
]

% ------------------------------------------------------------
% Main vertical nodes
% ------------------------------------------------------------

\node[inputblock] (input) {
    \textbf{Input}\\
    $(B,2,K,L)$
};

\node[convblock, below=of input] (inputproj) {
    \textbf{input\_projection}\\
    Conv1d, kernel size $1$
};

\node[resblock, below=of inputproj] (residual) {
    \textbf{ResidualBlock}\\
    with skip taps
};

\node[skipblock, below=0.9cm of residual] (sumskip) {
    $\displaystyle \sum \mathrm{skips}/\sqrt{n}$
};

\node[convblock, below=of sumskip] (outputproj) {
    \textbf{output\_projection}\\
    Conv1d, kernel size $1$
};

\node[opblock, below=of outputproj] (reshape) {
    \textbf{Reshape}
};

\node[inputblock, below=of reshape] (output) {
    \textbf{Output}\\
    $(B,K,L)$
};

% ------------------------------------------------------------
% Shape labels
% ------------------------------------------------------------

\node[shapetext, right=0.45cm of inputproj] {
    $(B,\mathrm{ch},K,L)$
};

% ResidualBlock dimension label moved upward to avoid overlap with ×n
\node[shapetext] at ($(residual.east)+(1.25,0.45)$) {
    $(B,\mathrm{ch},K{\times}L)$
};

% Skip aggregation dimension label moved slightly upward
\node[shapetext] at ($(sumskip.east)+(1.0,0.25)$) {
    $(B,\mathrm{ch},K{\times}L)$
};

\node[shapetext, right=0.45cm of outputproj] {
    $(B,1,K{\times}L)$
};

\node[shapetext, right=0.45cm of reshape] {
    $(B,K,L)$
};

% ------------------------------------------------------------
% Main arrows
% ------------------------------------------------------------

\draw[mainarrow] (input) -- (inputproj);
\draw[mainarrow] (inputproj) -- (residual);
\draw[mainarrow] (residual) -- (sumskip);
\draw[mainarrow] (sumskip) -- (outputproj);
\draw[mainarrow] (outputproj) -- (reshape);
\draw[mainarrow] (reshape) -- (output);

% ------------------------------------------------------------
% Conditioning inputs (mirror of architecture.tex layout)
% ------------------------------------------------------------

% Left side: diffusion time embedding
\node[projblock, left=2.2cm of residual] (diffproj) {
    \textbf{diffusion\_projection}\\
    Linear $(\mathrm{demb\_dim}\to \mathrm{ch})$
};

\node[inputblock, above=0.35cm of diffproj] (cnoise) {
    \textbf{$c_{\mathrm{noise}}$}\\
    $(B,\mathrm{demb\_dim})$
};

% Right side (top): feature/time conditioning
\node[projblock, right=2.2cm of residual] (condproj) {
    \textbf{cond\_projection}\\
    Conv1d $(193\to2\mathrm{ch})$, $k=1$
};

\node[inputblock, above=0.35cm of condproj] (sideinfo) {
    \textbf{side\_info}\\
    $(B,\mathrm{d_{temb}} + \mathrm{d_{femb}+1})$
};

% Right side (bottom): sinusoidal positional encoding
\node[inputblock, below=0.55cm of condproj] (sinpe) {
    \textbf{Sinusoidal PE}\\
    $(B,\mathrm{ch},L)$
};

% Arrows: conditioning inputs into ResidualBlock
\draw[sidearrow] (cnoise) -- (diffproj);
\draw[sidearrow] (diffproj.east) -- node[above, font=\scriptsize] {$(B,\mathrm{ch},K{\times}L)$} (residual.west);

\draw[sidearrow] (sideinfo) -- (condproj);
\draw[sidearrow] (condproj.west) -- node[above, font=\scriptsize] {$(B,2\mathrm{ch},K{\times}L)$} (residual.east);

% sinpe: go left to residual.east x, then up to residual.east
\draw[sidearrow] (sinpe.west) |- (residual.east);

% ------------------------------------------------------------
% Repeated-block annotation
% ------------------------------------------------------------

\node[anchor=west, font=\large] at ($(residual.east)+(0.75,0)$) {
    $\times n$
};

% ------------------------------------------------------------
% Skip-tap illustration
% ------------------------------------------------------------

% The skip arrow starts exactly from the exterior of the ResidualBlock
\coordinate (skipstart) at (residual.south east);

% Horizontal route point to the right of the ResidualBlock
\coordinate (skipright) at ($(skipstart)+(2.35,0)$);

% Vertical route point aligned with the exterior of the skip-sum block
\coordinate (skipdown) at (skipright |- sumskip.east);

% The arrow touches the exterior of the skip-sum block at its right edge
\draw[skiparrow]
    (skipstart)
    -- (skipright)
    -- node[midway, right=0.15cm, font=\scriptsize, align=center] {
        skip outputs
    }
    (skipdown)
    -- (sumskip.east);

% ------------------------------------------------------------
% Dashed container around the repeated residual block
% ------------------------------------------------------------

\node[
    draw=black,
    dashed,
    rounded corners=4pt,
    inner sep=0.25cm,
    fit=(residual)(cnoise)(diffproj)(sideinfo)(condproj)(sinpe)
] (rescontainer) {};

\end{tikzpicture}
    \caption{High-level overview of the CSDI denoising network. The two-channel input is projected into a latent space, processed by $n$ stacked residual blocks with skip connections, and decoded back to the predicted clean signal.}
